  \par\medskip\noindent\textbf{#1}\par\nobreak

% generated "Tables" and dash for this manuscript's one compressed range,

  \crefname{table}{Table}{Tables}%
  \crefrangeformat{table}{Tables~##3##1##4--##5##2##6}%
  \cref{#1}%
}}

}
