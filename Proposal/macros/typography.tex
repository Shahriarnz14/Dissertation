% Reduce only the normal body size by 1pt. Retain the approved 14.5pt
% baseline and the class's existing heading, caption and small-size scales.
\let\ProposalOriginalNormalSize\normalsize
\renewcommand{\normalsize}{%
  \ProposalOriginalNormalSize\fontsize{11}{14.5}\selectfont}
% Exact 11pt (rather than the standard class's 10.95pt) needs an explicit
% math-size entry; retain the existing 8pt/6pt script sizes.
\DeclareMathSizes{11}{11}{8}{6}

% Undo the template's alias that forces indentation after every heading.
\def\@afterindentfalse{\let\if@afterindent\iffalse}
% Omit the saved indentation box before paragraph tagging inserts PDF nodes.
% The legacy after-heading lastbox removal alone cannot remove an indent
% once marked-content nodes intervene in the tagged-PDF paragraph hook.
\AddToHook{para/begin}{%
  \if@nobreak\if@afterindent\else\OmitIndent\fi\fi}

% Change only the visual block. Keep automatic counters, labels, bookmarks,
% contents entries, float suppression and the CMU chapter hooks intact.
% Reduce only Lenny's initial 50pt skip to one third; retain its 20pt/40pt
% frame-to-title and title-to-content skips for chapters and appendices.
\newlength{\ProposalChapterTopSkip}
\setlength{\ProposalChapterTopSkip}{\dimexpr50pt/3\relax}
\patchcmd{\@makechapterhead}{\vspace*{50\p@}}
  {\vspace*{\ProposalChapterTopSkip}}
  {}{\ProposalPatchFailure{chapter opening top skip}}
\patchcmd{\chapter}
  {\vspace*{1cm} {\raggedright\LARGE\textbf {Chapter \thesection}\par}
    \vspace{1cm}
    {\setstretch{1.05}\raggedright\Huge \bfseries #1\par}
    \vspace{1cm}
    \justifying}
  {\@makechapterhead{#1}}
  {}{\ProposalPatchFailure{Lenny chapter renderer}}
\newcommand{\ProposalMakeAppendixHead}[1]{%
  \begingroup
    \def\@chapapp{Appendix}%
    \def\thechapter{\appendixnum}% only the displayed glyph; counters unchanged
    \@mainmattertrue
    \@makechapterhead{#1}%
  \endgroup}
\patchcmd{\cmu@appendix}
  {\vspace*{1cm}
    {\raggedright\LARGE\bfseries Appendix \appendixnum\par}
    {\raggedright\LARGE\bfseries #1\par}
    \vspace{1cm}}
  {\ProposalMakeAppendixHead{#1}}
  {}{\ProposalPatchFailure{Lenny appendix renderer}}
\apptocmd{\chapter}{\@afterindentfalse\@afterheading}
  {}{\ProposalPatchFailure{unindented first chapter paragraph}}
\apptocmd{\cmu@appendix}{\@afterindentfalse\@afterheading}
  {}{\ProposalPatchFailure{unindented first appendix paragraph}}
% Wrapped section headings align left without stretched interword spaces.
\appto\@subsection@style{\raggedright}

% Migrated unnumbered headings bypass @startsection; give them the same
% first-paragraph behavior without editing any manuscript module.
\newcommand{\ProposalUnindentedAfterHeading}[1]{%
  \ifcsdef{#1}
    {\csletcs{ProposalOriginal#1}{#1}%
     \csdef{#1}##1{\csuse{ProposalOriginal#1}{##1}%
       \@afterindentfalse\@afterheading}}
    {\ProposalPatchFailure{unnumbered heading #1}}%
}
